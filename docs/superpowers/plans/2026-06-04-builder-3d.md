/* ══════════════════════════════════════════════
   CINEMATIC HAND ANIMATION — 5 phases
══════════════════════════════════════════════ */
function playHandAnimation() {
  if (bouquetItems.length === 0) return;

  // Phase 1: Camera zoom in (0–800ms)
  controls.autoRotate = false;
  const startPos  = camera.position.clone();
  const endPos    = new THREE.Vector3(0, 2, 4);
  const startTime = performance.now();

  function zoomIn() {
    const t    = Math.min((performance.now() - startTime) / 800, 1);
    const ease = 1 - Math.pow(1 - t, 3);
    camera.position.lerpVectors(startPos, endPos, ease);
    camera.lookAt(0, 1.5, 0);
    if (t < 1) requestAnimationFrame(zoomIn);
    else showOverlay();
  }
  requestAnimationFrame(zoomIn);
}

function showOverlay() {
  // Phase 2: Capture Three.js frame and show overlay (800–1600ms)
  const overlay    = document.getElementById('hand-overlay');
  const snapCanvas = document.getElementById('hand-bouquet-canvas');

  renderer.render(scene, camera);
  const ctx = snapCanvas.getContext('2d');
  ctx.clearRect(0, 0, snapCanvas.width, snapCanvas.height);
  ctx.drawImage(renderer.domElement, 0, 0, snapCanvas.width, snapCanvas.height);

  // Reset hand-scene transform from any previous run
  const handScene = document.getElementById('hand-scene');
  handScene.classList.remove('pushing');
  handScene.style.animation = 'none';

  // Reset hand-wrapper
  const wrapper = document.getElementById('hand-wrapper');
  wrapper.classList.remove('rising');

  // Reset SVG grip
  const svg = document.getElementById('hand-svg');
  svg.classList.remove('hand-grip');
  svg.classList.add('hand-open');

  overlay.style.display = 'flex';
  requestAnimationFrame(() => overlay.classList.add('active'));

  setTimeout(raiseHand, 400);
}

function raiseHand() {
  // Phase 3: Hand rises (1600–2800ms)
  const wrapper = document.getElementById('hand-wrapper');
  wrapper.classList.add('rising');

  setTimeout(() => {
    const svg = document.getElementById('hand-svg');
    svg.classList.remove('hand-open');
    svg.classList.add('hand-grip');
    setTimeout(pushToScreen, 600);
  }, 400);
}

function spawnParticles() {
  const overlay = document.getElementById('hand-overlay');
  const colors  = bouquetItems.map(i => i.color).filter(Boolean);
  for (let i = 0; i < 20; i++) {
    const p = document.createElement('div');
    p.className = 'hand-particle';
    p.style.cssText = [
      `left:${40 + Math.random() * 20}%`,
      `top:${30 + Math.random() * 20}%`,
      `background:${colors[i % colors.length] || '#e8175d'}`,
      `animation-delay:${(Math.random() * 0.3).toFixed(2)}s`,
      `--dx:${Math.round((Math.random() - 0.5) * 200)}px`,
      `--dy:${Math.round((Math.random() - 0.5) * 200)}px`,
    ].join(';');
    overlay.appendChild(p);
  }
}

function pushToScreen() {
  // Phase 4: Push bouquet toward viewer (2800–3800ms)
  const handScene = document.getElementById('hand-scene');
  handScene.classList.add('pushing');
  spawnParticles();
  setTimeout(finishAnimation, 1000);
}

function finishAnimation() {
  // Phase 5: Fade out and redirect (3800–4500ms)
  const overlay = document.getElementById('hand-overlay');
  overlay.classList.add('fading');

  setTimeout(() => {
    overlay.style.display = 'none';
    overlay.classList.remove('active', 'fading');
    overlay.querySelectorAll('.hand-particle').forEach(p => p.remove());

    // Reset camera
    camera.position.set(0, 3, 9);
    controls.target.set(0, 1.5, 0);
    controls.autoRotate = true;
    controls.update();

    const msg = buildWhatsAppMessage();
    window.open(`https://wa.me/994702011930?text=${encodeURIComponent(msg)}`, '_blank');
  }, 700);
}
